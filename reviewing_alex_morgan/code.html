import pandas as pd
import numpy as np

# Load dataset
def load_and_clean_data(filepath):
    """Loads raw sales data and performs initial cleaning."""
    try:
        df = pd.read_csv(filepath)
        
        # Drop duplicate rows
        df_cleaned = df.drop_duplicates()
        
        # Handle missing values in 'Revenue' column (fill with median)
        if 'Revenue' in df_cleaned.columns:
            median_rev = df_cleaned['Revenue'].median()
            df_cleaned['Revenue'].fillna(median_rev, inplace=True)
            
        return df_cleaned
        
    except Exception as e:
        print(f"Error processing file: {e}")
        return None

# Main execution block
if __name__ == "__main__":
    processed_data = load_and_clean_data("raw_sales_2023.csv")
    if processed_data is not None:
        print("Data cleaned successfully. Shape:", processed_data.shape)
